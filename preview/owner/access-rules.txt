rules_version = '2';
service cloud.firestore { match /databases/{database}/documents {
// LEGACY TESTING ACCESS: migrate before adding real student information.
match /users/{userId} {
 allow read: if true;
 allow write: if request.auth != null && request.auth.uid == userId;
}
match /friendRequests/{requestId} { allow read, write: if request.auth != null; }
match /friendships/{friendshipId} { allow read, write: if request.auth != null; }
// Add inside match /databases/{database}/documents, alongside existing matches.
// Grants no access to student data. Does not repair legacy testing rules.
function platformOwner() {
  return request.auth != null
    && get(/databases/$(database)/documents/platformOwners/$(request.auth.uid)).data.active == true;
}
match /platformOwners/{uid} {
  allow get: if request.auth != null && request.auth.uid == uid;
  allow list, write: if false; // Provision/revoke only with trusted Firebase console/admin access.
}
function validSchool(d) {
  return d.keys().hasAll(['name','status','license','revision','updatedAt','updatedBy','changeId'])
    && d.keys().hasOnly(['name','status','license','revision','updatedAt','updatedBy','changeId'])
    && d.name is string && d.name.size() > 0 && d.name.size() <= 100
    && d.status in ['pilot','active','paused','archived']
    && d.revision is int && d.revision > 0
    && d.updatedAt == request.time && d.updatedBy == request.auth.uid
    && d.changeId is string && d.changeId.size() > 0 && d.changeId.size() <= 100
    && d.license is map
    && d.license.keys().hasAll(['plan','seatLimit','startsOn','endsOn','courses'])
    && d.license.keys().hasOnly(['plan','seatLimit','startsOn','endsOn','courses'])
    && d.license.plan in ['pilot','school']
    && d.license.seatLimit is int && d.license.seatLimit >= 1 && d.license.seatLimit <= 100000
    && d.license.startsOn is timestamp && d.license.endsOn is timestamp
    && d.license.endsOn > d.license.startsOn
    && d.license.courses == ['algebra1'];
}
function schoolAudit(schoolId, d, previousRevision) {
  let eventPath = /databases/$(database)/documents/ownerAudit/$(d.changeId);
  let event = getAfter(eventPath).data;
  return !exists(eventPath) && event.schoolId == schoolId
    && event.actor == request.auth.uid && event.at == request.time
    && event.previousRevision == previousRevision && event.revision == d.revision
    && event.snapshot == d;
}
match /schools/{schoolId} {
  allow get: if platformOwner() || activeMember(schoolId);
  allow list: if platformOwner();
  allow create: if platformOwner() && validSchool(request.resource.data)
    && request.resource.data.revision == 1 && schoolAudit(schoolId, request.resource.data, 0);
  allow update: if platformOwner() && validSchool(request.resource.data)
    && request.resource.data.revision == resource.data.revision + 1
    && request.resource.data.license.seatLimit >= occupied(schoolId)
    && schoolAudit(schoolId, request.resource.data, resource.data.revision);
  allow delete: if false; // Archive, retain history.
}
match /ownerAudit/{eventId} {
  allow read: if platformOwner();
  allow create: if platformOwner()
    && request.resource.data.keys().hasOnly(['schoolId','actor','at','previousRevision','revision','snapshot'])
    && request.resource.data.actor == request.auth.uid && request.resource.data.at == request.time
    && request.resource.data.previousRevision is int
    && request.resource.data.revision == request.resource.data.previousRevision + 1
    && getAfter(/databases/$(database)/documents/schools/$(request.resource.data.schoolId)).data == request.resource.data.snapshot
    && request.resource.data.snapshot.changeId == eventId;
  allow update, delete: if false;
}

// School membership is separate from client-editable profiles.
function activeMember(schoolId) {
  return request.auth != null
    && exists(/databases/$(database)/documents/schools/$(schoolId)/members/$(request.auth.uid))
    && get(/databases/$(database)/documents/schools/$(schoolId)/members/$(request.auth.uid)).data.state == 'active';
}
function schoolManager(schoolId) {
  return platformOwner() || (activeMember(schoolId)
    && get(/databases/$(database)/documents/schools/$(schoolId)/members/$(request.auth.uid)).data.role == 'administrator');
}
function licenseOpen(schoolId) {
  let s = get(/databases/$(database)/documents/schools/$(schoolId)).data;
  return s.status in ['pilot','active'] && s.license.startsOn <= request.time
    && request.time < s.license.endsOn && 'algebra1' in s.license.courses;
}
function occupied(schoolId) {
  let p = /databases/$(database)/documents/schools/$(schoolId)/usage/seats;
  return exists(p) ? get(p).data.occupied : 0;
}
function studentSeat(d) {
  return d.role == 'student' && d.state == 'active' ? 1 : 0;
}
function validMembership(d) {
  return d.keys().hasAll(['label','role','state','revision','updatedAt','updatedBy','changeId'])
    && d.keys().hasOnly(['label','role','state','revision','updatedAt','updatedBy','changeId'])
    && d.label is string && d.label.size() > 0 && d.label.size() <= 80
    && d.role in ['administrator','teacher','student'] && d.state in ['active','removed']
    && d.revision is int && d.revision > 0 && d.updatedAt == request.time
    && d.updatedBy == request.auth.uid && d.changeId is string && d.changeId.size() > 0 && d.changeId.size() <= 100;
}
function memberAudit(schoolId, uid, d, previousRevision) {
  let p = /databases/$(database)/documents/schools/$(schoolId)/membershipAudit/$(d.changeId);
  let a = getAfter(p).data;
  return !exists(p) && a.memberUid == uid && a.actor == request.auth.uid
    && a.at == request.time && a.previousRevision == previousRevision && a.revision == d.revision && a.snapshot == d;
}
function seatChange(schoolId, uid, d, previousSeat) {
  let delta = studentSeat(d) - previousSeat;
  let p = /databases/$(database)/documents/schools/$(schoolId)/usage/seats;
  let next = getAfter(p).data;
  return delta == 0 || (next.memberUid == uid && next.changeId == d.changeId
    && next.occupied == occupied(schoolId) + delta);
}
match /schools/{schoolId}/members/{uid} {
  allow get: if request.auth != null && (request.auth.uid == uid || schoolManager(schoolId));
  allow list: if schoolManager(schoolId);
  allow create: if schoolManager(schoolId) && validMembership(request.resource.data)
    && request.resource.data.revision == 1
    && (platformOwner() || (uid != request.auth.uid && request.resource.data.role != 'administrator'))
    && (request.resource.data.state == 'removed' || licenseOpen(schoolId))
    && memberAudit(schoolId, uid, request.resource.data, 0)
    && seatChange(schoolId, uid, request.resource.data, 0);
  allow update: if schoolManager(schoolId) && validMembership(request.resource.data)
    && request.resource.data.revision == resource.data.revision + 1
    && (platformOwner() || (uid != request.auth.uid && resource.data.role != 'administrator' && request.resource.data.role != 'administrator'))
    && (request.resource.data.state == 'removed' || licenseOpen(schoolId))
    && memberAudit(schoolId, uid, request.resource.data, resource.data.revision)
    && seatChange(schoolId, uid, request.resource.data, studentSeat(resource.data));
  allow delete: if false;
}
function validSeatWrite(schoolId) {
  let d = request.resource.data;
  let memberPath = /databases/$(database)/documents/schools/$(schoolId)/members/$(d.memberUid);
  let before = exists(memberPath) ? studentSeat(get(memberPath).data) : 0;
  let after = getAfter(memberPath).data;
  return d.keys().hasAll(['occupied','memberUid','changeId','updatedAt'])
    && d.keys().hasOnly(['occupied','memberUid','changeId','updatedAt'])
    && d.updatedAt == request.time && d.occupied is int && d.occupied >= 0
    && d.occupied <= getAfter(/databases/$(database)/documents/schools/$(schoolId)).data.license.seatLimit
    && d.occupied == occupied(schoolId) + studentSeat(after) - before
    && after.changeId == d.changeId && after.updatedAt == request.time
    && (!exists(memberPath) || get(memberPath).data.changeId != after.changeId);
}
match /schools/{schoolId}/usage/seats {
  allow read: if schoolManager(schoolId);
  allow create, update: if schoolManager(schoolId) && validSeatWrite(schoolId);
  allow delete: if false;
}
match /schools/{schoolId}/membershipAudit/{eventId} {
  allow read: if schoolManager(schoolId);
  allow create: if schoolManager(schoolId)
    && request.resource.data.keys().hasOnly(['memberUid','actor','at','previousRevision','revision','snapshot'])
    && request.resource.data.actor == request.auth.uid && request.resource.data.at == request.time
    && request.resource.data.previousRevision is int
    && request.resource.data.revision == request.resource.data.previousRevision + 1
    && getAfter(/databases/$(database)/documents/schools/$(schoolId)/members/$(request.resource.data.memberUid)).data == request.resource.data.snapshot
    && request.resource.data.snapshot.changeId == eventId;
  allow update, delete: if false;
}

}}
